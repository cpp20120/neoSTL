cmake_minimum_required(VERSION 3.20)
project(neo_stl_poc_v9 LANGUAGES CXX)

option(NEO_SANITIZERS "Enable ASan/UBSan for GCC/Clang" OFF)
add_library(neo INTERFACE)
target_include_directories(neo INTERFACE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(neo INTERFACE cxx_std_23)

add_executable(neo_demo demo.cpp)
add_executable(neo_v3_tests v3_tests.cpp)
add_executable(neo_v4_tests v4_tests.cpp)
add_executable(neo_lifetime_tests lifetime_tests.cpp)
add_executable(neo_exception_tests exception_tests.cpp)
add_executable(neo_iterator_contract_tests iterator_contract_tests.cpp)
add_executable(neo_concepts_tests concepts_tests.cpp)
add_executable(neo_language_features_tests language_features_tests.cpp)
add_executable(neo_ranges_to_tests ranges_to_tests.cpp)
add_executable(neo_policy_pack_tests policy_pack_tests.cpp)
add_executable(neo_presets_tests presets_tests.cpp)
add_executable(neo_adaptor_closure_tests adaptor_closure_tests.cpp)
add_executable(neo_range_capabilities_tests range_capabilities_tests.cpp)
add_executable(neo_cpo_dispatch_tests cpo_dispatch_tests.cpp)
target_link_libraries(neo_demo PRIVATE neo)
target_link_libraries(neo_v3_tests PRIVATE neo)
target_link_libraries(neo_v4_tests PRIVATE neo)
target_link_libraries(neo_lifetime_tests PRIVATE neo)
target_link_libraries(neo_exception_tests PRIVATE neo)
target_link_libraries(neo_iterator_contract_tests PRIVATE neo)
target_link_libraries(neo_concepts_tests PRIVATE neo)
target_link_libraries(neo_language_features_tests PRIVATE neo)
target_link_libraries(neo_ranges_to_tests PRIVATE neo)
target_link_libraries(neo_policy_pack_tests PRIVATE neo)
target_link_libraries(neo_presets_tests PRIVATE neo)
target_link_libraries(neo_adaptor_closure_tests PRIVATE neo)
target_link_libraries(neo_range_capabilities_tests PRIVATE neo)
target_link_libraries(neo_cpo_dispatch_tests PRIVATE neo)
if(MSVC)
    foreach(target IN ITEMS neo_demo neo_v3_tests neo_v4_tests neo_lifetime_tests neo_exception_tests neo_iterator_contract_tests neo_concepts_tests neo_language_features_tests neo_ranges_to_tests neo_policy_pack_tests neo_presets_tests neo_adaptor_closure_tests neo_range_capabilities_tests neo_cpo_dispatch_tests)
        target_compile_options(${target} PRIVATE /UNDEBUG /W4 /WX /permissive-)
    endforeach()
else()
    foreach(target IN ITEMS neo_demo neo_v3_tests neo_v4_tests neo_lifetime_tests neo_exception_tests neo_iterator_contract_tests neo_concepts_tests neo_language_features_tests neo_ranges_to_tests neo_policy_pack_tests neo_presets_tests neo_adaptor_closure_tests neo_range_capabilities_tests neo_cpo_dispatch_tests)
        target_compile_options(${target} PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
        if(NEO_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endforeach()
endif()
enable_testing()
add_test(NAME neo_demo COMMAND neo_demo)
add_test(NAME neo_v3_tests COMMAND neo_v3_tests)

add_test(NAME neo_v4_tests COMMAND neo_v4_tests)

add_test(NAME neo_lifetime_tests COMMAND neo_lifetime_tests)

add_test(NAME neo_presets_tests COMMAND neo_presets_tests)

add_test(NAME neo_adaptor_closure_tests COMMAND neo_adaptor_closure_tests)
add_test(NAME neo_range_capabilities_tests COMMAND neo_range_capabilities_tests)

add_test(NAME neo_cpo_dispatch_tests COMMAND neo_cpo_dispatch_tests)
